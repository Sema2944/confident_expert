from __future__ import annotations


def collect_outfit_file_ids(items_payload: dict[str, list[str]]) -> list[str]:
    category_order = ["top", "bottom", "dress", "outerwear", "shoes", "accessories"]
    file_ids: list[str] = []
    for category in category_order:
        file_ids.extend(items_payload.get(category, []))
    return file_ids


def items_payload_signature(items_payload: dict[str, list[str]]) -> tuple[tuple[str, tuple[str, ...]], ...]:
    normalized = []
    for category in sorted(items_payload):
        normalized.append((category, tuple(items_payload.get(category, []))))
    return tuple(normalized)
